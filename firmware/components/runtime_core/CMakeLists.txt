set(RUNTIME_CORE_SOURCES runtime_config.c http_exchange.c ota_manifest.c ui_snapshot.c wifi_provisioning_core.c improv_serial_core.c)
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${RUNTIME_CORE_SOURCES} INCLUDE_DIRS include REQUIRES glucose_core)
else()
    cmake_minimum_required(VERSION 3.24)
    project(runtime_core LANGUAGES C)
    option(RUNTIME_CORE_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
    option(RUNTIME_CORE_HOST_TEST "Enable host-only test behavior" OFF)
    option(RUNTIME_ALLOW_IP_LITERAL "Allow IP literals in host tests" OFF)
    add_subdirectory(../glucose_core glucose_core-build)
    add_library(runtime_core STATIC ${RUNTIME_CORE_SOURCES})
    target_include_directories(runtime_core PUBLIC include)
    target_link_libraries(runtime_core PUBLIC glucose_core)
    add_executable(runtime_core_tests test/test_runtime_core.c)
    target_link_libraries(runtime_core_tests PRIVATE runtime_core)
    add_executable(improv_serial_core_tests test/test_improv_serial_core.c)
    target_link_libraries(improv_serial_core_tests PRIVATE runtime_core)
    add_executable(ui_task_host_tests test/ui_task_host.c test/test_ui_task_host.c ../../main/ui_task.c)
    target_include_directories(ui_task_host_tests PRIVATE test/fakes ../../main test)
    target_link_libraries(ui_task_host_tests PRIVATE runtime_core)
    foreach(target runtime_core runtime_core_tests improv_serial_core_tests ui_task_host_tests)
        set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
        if(RUNTIME_CORE_HOST_TEST)
            target_compile_definitions(${target} PRIVATE RUNTIME_CORE_HOST_TEST)
        endif()
        if(RUNTIME_ALLOW_IP_LITERAL)
            target_compile_definitions(${target} PRIVATE RUNTIME_ALLOW_IP_LITERAL)
        endif()
    endforeach()
    if(RUNTIME_CORE_ENABLE_SANITIZERS)
        foreach(target runtime_core runtime_core_tests improv_serial_core_tests ui_task_host_tests)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
        endforeach()
        foreach(target runtime_core_tests improv_serial_core_tests ui_task_host_tests)
            target_link_options(${target} PRIVATE
                -fsanitize=address,undefined -fno-sanitize-recover=all)
        endforeach()
    endif()
    enable_testing()
    add_test(NAME runtime_core COMMAND runtime_core_tests)
    add_test(NAME improv_serial_core COMMAND improv_serial_core_tests)
    add_test(NAME ui_task_host COMMAND ui_task_host_tests)
    if(RUNTIME_CORE_ENABLE_SANITIZERS)
        # ui_task is a process-lifetime FreeRTOS owner with no stop API; the
        # host fake intentionally leaves its queue/task allocations live at exit.
        set_tests_properties(ui_task_host PROPERTIES
            ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
    endif()
endif()
